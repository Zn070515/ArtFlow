# M15.3 Attendance / Check-in Implementation Plan

> **For agentic workers:** Execute inline in this session. The repository rule in
> `AGENTS.md` forbids spawning subagents. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Add the optional unified `attendance` capability for participant,
ticket, and staff check-in at named activity points, with session-scoped,
mode-partitioned, append-only facts and Ops/Live read surfaces.

**Architecture:** A new capability under `packages/domain`, four PostgreSQL
tables (points, staff roster, facts, current projection), a runtime service that
writes check-in/undo through the single command path, a protected Fastify route,
and an Ops card plus a Live summary. Check-in and undo require a running
`checkin` execution session; ticket check-in never changes ticket state.

**Tech Stack:** TypeScript 7, Kysely/PostgreSQL 18, Fastify 5, TypeBox,
React 19, TanStack Query, Vitest, Playwright.

**Spec:** [2026-11-10-attendance-design.md](../../specs/2026-11-10-attendance-design.md)

```ts
import { capabilityDefinitionHash } from "../compiler.js";
import type {
  CapabilityCompileResult,
  CapabilityCompiler,
  CapabilityDefinition,
  CompiledCapabilityPlan,
} from "../model.js";
import { deepFreeze } from "../model.js";
import type { CapabilityRegistration } from "../registry.js";

export type AttendanceSubjectType = "participant" | "ticket" | "staff";

export interface CompiledAttendanceArtifact {
  readonly schemaVersion: 1;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

export function createEmptyAttendanceDefinition(): CapabilityDefinition {
  return { schemaVersion: 1, payload: { schemaVersion: 1 } };
}

export function compileAttendanceDefinition(
  definition: CapabilityDefinition,
): CapabilityCompileResult {
  if (definition.schemaVersion !== 1 || !isRecord(definition.payload)) {
    return deepFreeze({
      ok: false,
      issues: [{ code: "PAYLOAD_INVALID", message: "Not an attendance definition.", path: "payload" }],
    });
  }
  if (definition.payload.schemaVersion !== 1) {
    return deepFreeze({
      ok: false,
      issues: [{ code: "SCHEMA_VERSION", message: "Only schemaVersion 1 is supported.", path: "payload.schemaVersion" }],
    });
  }
  const unknown = Object.keys(definition.payload).filter((key) => key !== "schemaVersion");
  if (unknown.length > 0) {
    return deepFreeze({
      ok: false,
      issues: [{
        code: "PAYLOAD_UNKNOWN_KEY",
        message: `Attendance definitions only accept schemaVersion; found ${unknown.join(", ")}.`,
        path: `payload.${unknown[0]}`,
      }],
    });
  }
  const artifact: CompiledAttendanceArtifact = { schemaVersion: 1 };
  const plan: CompiledCapabilityPlan = {
    ok: true,
    capabilityKey: "attendance",
    contentHash: capabilityDefinitionHash(definition),
    artifact: deepFreeze(artifact),
  };
  return deepFreeze(plan);
}

export const attendanceCompiler: CapabilityCompiler = {
  capabilityKey: "attendance",
  compile: compileAttendanceDefinition,
};

export const ATTENDANCE_COMMAND_SCOPES: Readonly<Record<string, readonly string[]>> = deepFreeze({
  "attendance.checkin": ["checkin"],
  "attendance.undo": ["checkin"],
});

export const attendanceCapability: CapabilityRegistration = deepFreeze({
  manifest: {
    identity: "attendance",
    version: "1",
    dependencies: [],
    conflicts: [],
    definitionSchemaVersion: 1,
    commands: [
      "attendance.point.create",
      "attendance.point.update",
      "attendance.staff.add",
      "attendance.staff.update",
      "attendance.checkin",
      "attendance.undo",
    ],
    queries: ["attendance.list", "attendance.staff.list"],
    events: [
      "AttendancePointChanged",
      "AttendanceStaffChanged",
      "AttendanceCheckedIn",
      "AttendanceUndone",
    ],
    permissions: ["attendance.manage"],
    surfaces: ["studio", "ops", "live"],
    scopes: ["checkin"],
    commandScopes: ATTENDANCE_COMMAND_SCOPES,
  },
  compiler: attendanceCompiler,
  emptyDefinition: createEmptyAttendanceDefinition,
});
```

```ts
export async function getAttendancePointForUpdate(
  trx: Transaction<Database>,
  activityId: string,
  key: string,
): Promise<StoredAttendancePoint | null> {
  const row = await trx
    .selectFrom("attendance_point")
    .selectAll()
    .where("activity_id", "=", activityId)
    .where("key", "=", key)
    .forUpdate()
    .executeTakeFirst();
  return row === undefined ? null : rowToPoint(row);
}

export async function updateAttendancePoint(
  trx: Transaction<Database>,
  input: {
    readonly activityId: string;
    readonly id: string;
    readonly expectedRevision: number;
    readonly name: string;
    readonly subjectTypes: readonly AttendanceSubjectType[];
    readonly active: boolean;
  },
): Promise<StoredAttendancePoint> {
  const row = await trx
    .updateTable("attendance_point")
    .set({
      name: input.name,
      subject_types: JSON.stringify(input.subjectTypes),
      active: input.active,
      revision: input.expectedRevision + 1,
      updated_at: new Date(),
    })
    .where("activity_id", "=", input.activityId)
    .where("id", "=", input.id)
    .where("revision", "=", input.expectedRevision)
    .returningAll()
    .executeTakeFirst();
  if (row === undefined) throw new StaleRevisionError(input.id);
  return rowToPoint(row);
}

    .selectAll()

    .orderBy("key")
    .execute();
  return rows.map(rowToPoint);
}
```

```ts
async function requireEligibleSubject(
  trx: import("kysely").Transaction<Database>,
  activityId: string,
  subjectType: AttendanceSubjectType,
  subjectId: string,
): Promise<void> {
  if (subjectType === "participant") {
    const registration = await getRegistration(trx, subjectId);
    if (registration === null || registration.activityId !== activityId) {
      throw new DomainError("CROSS_ACTIVITY", "The registration belongs to another activity.");
    }
    if (registration.status !== "approved") {
      throw new DomainError("SUBJECT_NOT_ELIGIBLE", "Only an approved registration can be checked in.");
    }
    return;
  }
  if (subjectType === "ticket") {
    const ticket = await getTicket(trx, subjectId);
    if (ticket === null || ticket.activityId !== activityId) {
      throw new DomainError("CROSS_ACTIVITY", "The ticket belongs to another activity.");
    }
    if (ticket.state === "revoked" || ticket.state === "expired") {
      throw new DomainError("SUBJECT_NOT_ELIGIBLE", "A revoked or expired ticket cannot be checked in.");
    }
    return;
  }
  const staff = await getAttendanceStaff(trx, activityId, subjectId);
  if (staff === null || !staff.active) {
    throw new DomainError("SUBJECT_NOT_ELIGIBLE", "The staff roster entry is not active.");
  }
}

export async function checkIn(
  db: import("kysely").Kysely<Database>,
  deps: RuntimeDeps,
  input: AttendanceCheckinInput,
): Promise<CommandOutcome<StoredAttendanceCurrent>> {
  const actor = input.actor;
  const eventDeps = withActor(deps, actor);
  const note = normalizeNote(input.note);
  const subjectType = requireSubjectType(input.subjectType);
  return executeCommand(db, {
    commandId: input.commandId,
    activityId: input.activityId,
    capability: "attendance",
    commandType: "attendance.checkin",
    scope: "checkin",
    actor,
    payload: { pointKey: input.pointKey, subjectType, subjectId: input.subjectId, note },
  }, async (trx) => {
    requirePermission(actor, "attendance.manage");
    const activity = await getActivityForUpdate(trx, input.activityId);
    if (activity === null) throw new DomainError("NOT_FOUND", "The activity was not found.");
    requireActive(activity);
    requireAttendanceEnabled(activity);
    const { session } = await requireRunningCommandSession(trx, activity, "attendance.checkin", "checkin");
    if (session.mode !== "formal" && session.mode !== "test") {
      throw new DomainError("INVALID_MODE", "Attendance requires a formal or test check-in session.");
    }
    const point = await getAttendancePointForUpdate(trx, input.activityId, input.pointKey);
    if (point === null) throw new DomainError("POINT_NOT_FOUND", "The attendance point was not found.");
    if (!point.active) throw new DomainError("POINT_INACTIVE", "The attendance point is inactive.");
    if (!point.subjectTypes.includes(subjectType)) {
      throw new DomainError("SUBJECT_TYPE_NOT_ALLOWED", "This point does not accept that subject type.");
    }
    await requireEligibleSubject(trx, input.activityId, subjectType, input.subjectId);
    const current = await getAttendanceCurrentForUpdate(trx, {
      activityId: input.activityId,
      mode: session.mode,
      pointKey: input.pointKey,
      subjectType,
      subjectId: input.subjectId,
    });
    if (current !== null && current.status === "present") {
      throw new DomainError("ALREADY_CHECKED_IN", "The subject is already checked in at this point.");
    }
    const now = deps.now();
    await insertAttendanceFact(trx, {
      id: idFromCommand("attendance-fact", input.commandId),
      activityId: input.activityId,
      mode: session.mode,
      pointKey: input.pointKey,
      subjectType,
      subjectId: input.subjectId,
      action: "checkin",
      note,
      actorId: actor.actorId,
      commandId: input.commandId,
      occurredAt: now,
    });
    const saved = await upsertAttendanceCurrent(trx, {
      activityId: input.activityId,
      mode: session.mode,
      pointKey: input.pointKey,
      subjectType,
      subjectId: input.subjectId,
      status: "present",
      note,
      revision: (current?.revision ?? 0) + 1,
      updatedAt: now,
    });
    await appendEngineEvent(trx, eventDeps, {
      type: "AttendanceCheckedIn",
      activityId: input.activityId,
      aggregateType: "attendance",
      aggregateId: `${input.pointKey}:${subjectType}:${input.subjectId}`,
      revision: saved.revision,
      capability: "attendance",
      payload: { pointKey: input.pointKey, subjectType, subjectId: input.subjectId },
    });
    return { body: saved };
  });
```

```ts
  const answers = typeof row.answers === "string" ? JSON.parse(row.answers) as Record<string, unknown> : row.answers;
  const name = answers !== null && typeof answers === "object" ? (answers as Record<string, unknown>).name : undefined;

export async function listAttendance(
  db: import("kysely").Kysely<Database>,
  deps: RuntimeDeps,
  input: AttendanceListInput,
): Promise<AttendanceListView> {
  const actor = runtimeActor(deps);
  requireAttendanceRead(actor);
  const activity = await getActivity(db, input.activityId);
  if (activity === null) throw new DomainError("NOT_FOUND", "The activity was not found.");
  requireAttendanceEnabled(activity);
  const mode = input.mode ?? "formal";
  const points = await listAttendancePoints(db, input.activityId);
  const staff = await listAttendanceStaff(db, input.activityId);
  const current = await listAttendanceCurrent(db, input.activityId, mode);
  const currentByKey = new Map(current.map((row) => [
    `${row.pointKey}\u0000${row.subjectType}\u0000${row.subjectId}`,
    row,
  ]));
  const registrations = await listRegistrations(db, input.activityId);
  const tickets = await listTicketsForAttendance(db, input.activityId);

  const rows: AttendanceRowView[] = [];
  for (const point of points) {
    if (input.point !== undefined && point.key !== input.point) continue;
    if (point.subjectTypes.includes("participant")) {
      for (const registration of registrations) {
        const eligible = registration.status === "approved";
        const row = currentByKey.get(`${point.key}\u0000participant\u0000${registration.id}`);
        if (!eligible && row === undefined) continue;
        rows.push({
          pointKey: point.key,
          subjectType: "participant",
          subjectId: registration.id,
          label: participantLabel(registration),
          eligible,
          status: row?.status ?? "not_checked_in",
          note: row?.note ?? null,
          updatedAt: row?.updatedAt ?? null,
        });
      }
    }
    if (point.subjectTypes.includes("ticket")) {
      for (const ticket of tickets) {
        const eligible = ticket.state === "issued" || ticket.state === "redeemed";
        const row = currentByKey.get(`${point.key}\u0000ticket\u0000${ticket.id}`);
        if (!eligible && row === undefined) continue;
        rows.push({
          pointKey: point.key,
          subjectType: "ticket",
          subjectId: ticket.id,
          label: ticket.publicCode,
          eligible,
          status: row?.status ?? "not_checked_in",
          note: row?.note ?? null,
          updatedAt: row?.updatedAt ?? null,
        });
      }
    }
    if (point.subjectTypes.includes("staff")) {
      for (const entry of staff) {
        const eligible = entry.active;
        const row = currentByKey.get(`${point.key}\u0000staff\u0000${entry.id}`);
        if (!eligible && row === undefined) continue;
        rows.push({
          pointKey: point.key,
          subjectType: "staff",
          subjectId: entry.id,
          label: entry.displayName,
          eligible,
          status: row?.status ?? "not_checked_in",
          note: row?.note ?? null,
          updatedAt: row?.updatedAt ?? null,
        });
      }
    }
  }

  const scoped = rows
    .filter((row) => input.subjectType === undefined || row.subjectType === input.subjectType)
    .sort((left, right) =>
      left.pointKey.localeCompare(right.pointKey) ||
      left.subjectType.localeCompare(right.subjectType) ||
      left.label.localeCompare(right.label) ||
      left.subjectId.localeCompare(right.subjectId),
    );
  const query = input.query?.trim().toLowerCase() ?? "";
  const filtered = scoped.filter(
    (row) => query.length === 0 || row.label.toLowerCase().includes(query) || row.subjectId.toLowerCase().includes(query),
  );

  const summary: AttendanceSummaryView[] = (["participant", "ticket", "staff"] as const).map((subjectType) => {
    const eligibleRows = scoped.filter((row) => row.subjectType === subjectType && row.eligible);
    return {
      subjectType,
      expected: eligibleRows.length,
      present: eligibleRows.filter((row) => row.status === "present").length,
      undone: eligibleRows.filter((row) => row.status === "undone").length,
    };
  });

  const limit = Math.min(Math.max(input.limit ?? 200, 1), 500);
  const offset = Math.max(input.offset ?? 0, 0);
  return {
    activityId: input.activityId,
    mode,
    points,
    staff,
    rows: filtered.slice(offset, offset + limit),
    summary,
  };
}
```

```ts
export async function resolveTicketCredentialForAttendance(
  db: Kysely<Database>,
  deps: RuntimeDeps,
  rawCredential: string,
): Promise<AttendanceTicketResolution> {
  const signingSecret = secret(deps);
  const parsed = parseCredential(rawCredential, "T", signingSecret);
  if (parsed === null) throw new DomainError("INVALID_TICKET", "The ticket credential is invalid.");
  const candidate = await findTicketByCredential(db, parsed.publicCode, parsed.version);
  if (
    candidate === null ||
    !validCredentialForRow(rawCredential, "T", candidate.publicCode, candidate.credentialVersion, signingSecret)
  ) {
    throw new DomainError("INVALID_TICKET", "The ticket credential is invalid.");
  }
  return { ticketId: candidate.id, activityId: candidate.activityId };
}
```
